How to wire up a free Google Sheet backend
  1. Create a new Google Sheet with columns: timestamp, worker_name, worker_id, action, lat, lng, accuracy_m, distance_m, in_fence, site, device_id, client_time.
  2. In the sheet: Extensions → Apps Script. Paste the snippet below.
  3. Deploy → New deployment → Web app. Access: Anyone. Copy the URL and paste it here.
const SHEET_ID = 'PASTE_YOUR_SHEET_ID_HERE';

function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  const sheet = SpreadsheetApp.openById(SHEET_ID).getSheets()[0];
  sheet.appendRow([
    new Date(),           // server timestamp
    data.worker_name,
    data.worker_id,
    data.action,
    data.lat, data.lng, data.accuracy_m,
    data.distance_m, data.in_fence,
    data.site, data.device_id, data.client_time
  ]);
  return ContentService
    .createTextOutput(JSON.stringify({ok:true}))
    .setMimeType(ContentService.MimeType.JSON);
}
